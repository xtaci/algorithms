cmake_minimum_required(VERSION 3.5)
project(algorithm LANGUAGES CXX)

include_directories( "include" )
include_directories( "." )
if( MSVC )
	include_directories( "msvc" )
endif()

#add_executable(word_seg "src/word_seg_demo.cpp")

file( GLOB APP_SOURCES src/*.cpp )
foreach( appsourcefile ${APP_SOURCES} )

	if( MSVC )
		target_compile_options( ${demo_name} PRIVATE "/FI${CMAKE_CURRENT_SOURCE_DIR}/msvc/alg_vs.h" )
	endif()
endforeach( appsourcefile ${APP_SOURCES} )
